cmake_minimum_required(VERSION 3.20)
project(estudos LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_executable(estudos main.cpp)
target_compile_options(estudos PRIVATE -Wall -Wextra -Wpedantic -Wshadow)
option(ENABLE_SANITIZERS "Detectar erros de memoria e comportamento indefinido" OFF)
if(ENABLE_SANITIZERS)
  target_compile_options(estudos PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(estudos PRIVATE -fsanitize=address,undefined)
endif()
